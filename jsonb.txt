1) conversione jsonb
    ALTER TABLE db_social_platform.posts
    ALTER COLUMN tags TYPE jsonb;

2) Recuperare tutti i media che hanno il tag "relax".
    with post_relax as(
        SELECT * FROM db_social_platform.posts p
        where  ( tags @> '["relax"]' )
    )

    SELECT m.*,p.tags  FROM db_social_platform.medias m
    inner join db_social_platform.media_post mp on m.id = mp.media_id
    inner join post_relax p on p.id=mp.post_id

3) BONUS
    CREATE VIEW db_social_platform.v_tags AS
    SELECT DISTINCT
        t.tag
    FROM db_social_platform.posts p
    CROSS JOIN LATERAL jsonb_array_elements_text(p.tags) AS t(tag)
    WHERE p.tags IS NOT NULL;
